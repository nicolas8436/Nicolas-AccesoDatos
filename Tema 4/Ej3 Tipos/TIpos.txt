--Ejercicio 1
CREATE TYPE TELEFONOS AS VARRAY(3) OF VARCHAR2(15);

--Ejercicio 2
CREATE TYPE DIRECCION AS OBJECT (
    CALLE VARCHAR2(30),
    POBLACION VARCHAR2(30),
    CODPOSTAL VARCHAR2(30),
    PROVINCIA VARCHAR2(30)
)

CREATE TYPE CLIENTE AS OBJECT (
    ID_CLIENTE VARCHAR2(30),
    NOMBRE VARCHAR2(30),
    NIF VARCHAR2(30),
    DIR DIRECCION,
    TLF TELEFONOS
)

CREATE TYPE PRODUCTO AS OBJECT(
    ID_PRODUCTO VARCHAR(30),
    DESCRIPCION VARCHAR(50),
    PVP NUMBER,
    STOCK_ACTUAL NUMBER
)

CREATE TYPE LINEA_VENTA AS OBJECT(
    NUM_LINEA NUMBER,
    ID_PROD REF PRODUCTO,
    CANTIDAD NUMBER
)

--Ejercicio 3
CREATE TYPE LINEAS_VENTAS AS TABLE OF LINEA_VENTA;

--Ejercicio 4
CREATE TYPE VENTA AS OBJECT(
    ID_VENTA NUMBER,
    ID_CLI REF CLIENTE,
    FECHAVENTA DATE,
    LINEAS LINEAS_VENTAS,
    
    MEMBER FUNCTION TOTAL_VENTAS RETURN NUMBER
)

--Ejercicio 5

CREATE OR REPLACE TYPE BODY VENTA AS 
MEMBER FUNCTION TOTAL_VENTAS RETURN NUMBER
IS
    TOTAL NUMBER := 0;
BEGIN
    FOR i IN 1..SELF.LINEAS.COUNT LOOP
    TOTAL := TOTAL + (SELF.LINEAS(i).ID_PROD.PVP * SELF.LINEAS(i).CANTIDAD);
    END LOOP;
        
    RETURN TOTAL;
    END;
END;

CREATE TABLE TABLA_CLIENTES OF CLIENTE(

CREATE TABLE TABLA_PRODUCTOS OF PRODUCTO(
    ID_PRODUCTO PRIMARY KEY
);

CREATE TABLE TABLA_VENTAS OF VENTA (
    ID_VENTA PRIMARY KEY
) NESTED TABLE LINEAS STORE AS LINEAS_VENTAS_TABLA
